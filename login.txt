    <script>
        checkAuth(); // Redirect jika sudah punya token

        document.getElementById('loginForm').addEventListener('submit', async function(e) {
            e.preventDefault();
            const btn = document.getElementById('btnLogin');
            btn.disabled = true;
            btn.innerHTML = '<div class="spinner"></div> Autentikasi...';

            const user = document.getElementById('username').value;
            const pass = document.getElementById('password').value;

            const res = await fetchAPI('login', { username: user, password: pass });

            if (res && res.success) {
                // Simpan sesi
                sessionStorage.setItem('sim_token', res.data.token);
                sessionStorage.setItem('sim_user_name', res.data.nama_lengkap);
                sessionStorage.setItem('sim_role', res.data.role);
                window.location.href = 'index.html';
            } else {
                showToast(res ? res.message : "Terjadi kesalahan", true);
                btn.disabled = false;
                btn.innerHTML = 'Masuk Aplikasi';
            }
        });
    </script>
